// ============================== TICKET ID ==============================
    public int ticketID()
    {
        ticketId++;
        name();
        return ticketId;
    }

    // ============================== NAME ==============================
    public String name()
    {
        System.out.println("Enter clients name: ");
        String clientName = scanner.nextLine();
        floor();
        return clientName;
    }

    // ============================== PROBLEM ==============================
    public String problem()
    {

        String problem = scanner.nextLine();

        printTicket(ticketId, problem, floor, problem, problem, problem);
        return problem;
    }

    public void printTicket(int ticketId, String clientName, int floor, String department, String division, String problem)
